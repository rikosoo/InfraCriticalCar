\fi

% a little stretch keeps long compound terms from overflowing narrow measures
\emergencystretch=1em

\newcommand{\capm}{\textsc{capm}}
\newcommand{\attck}{ATT\&CK}

% Generated tables size their text columns from these lengths and pick their
% font through \capmtablefont, so the same generated file fits a two-column
% IEEE page and a single-column Elsevier or SAE page.
\newlength{\capmpathcol}   \setlength{\capmpathcol}{11cm}
\newlength{\capmvictimcol} \setlength{\capmvictimcol}{4.2cm}
\newlength{\capmstdcol}    \setlength{\capmstdcol}{5cm}
\newlength{\capmnamecol}   \setlength{\capmnamecol}{3.6cm}
\newcommand{\capmtablefont}{\small}

% The architecture figure is the one that needs more width than a column: it
% spans both in a two-column layout and turns a landscape page in a single
% column one, where shrinking it to the measure would cost its legibility.
\newenvironment{capmwidefigure}{\begin{figure*}}{\end{figure*}}

% Single-column venues: the generated tables and the architecture figure use the
% starred float forms, which only exist in two-column layouts, and the wide
% columns have to shrink to the narrower text block.
\newcommand{\capmsinglecolumnfloats}{%
  \renewenvironment{table*}{\begin{table}}{\end{table}}%
  \renewenvironment{figure*}{\begin{figure}}{\end{figure}}%
  \renewenvironment{capmwidefigure}{\begin{sidewaysfigure}}{\end{sidewaysfigure}}%
  \setlength{\capmpathcol}{7.4cm}%
  \setlength{\capmvictimcol}{2.4cm}%
  \setlength{\capmstdcol}{3.2cm}%
  \setlength{\capmnamecol}{3.0cm}%
  \renewcommand{\capmtablefont}{\scriptsize}%
}
